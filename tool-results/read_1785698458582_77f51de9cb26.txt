   136→      }

   212→  // ═══ FIX: Serialize write operations to prevent race conditions ═══
   213→  // WARNING: This lock is NOT reentrant. Do NOT call removeFromQueue()
   214→  // or addToSyncQueue() from within a function that already holds the lock.
   215→  // syncPendingItems() uses _saveQueue() directly (not removeFromQueue) to avoid deadlock.
   216→  final _lockQueue = <Completer<void>>[];
   217→
   218→  Future<T> _withWriteLock<T>(Future<T> Function() action) async {
   219→    final prevLock = _lockQueue.isNotEmpty ? _lockQueue.last : null;
   220→    final myLock = Completer<void>();
   221→    _lockQueue.add(myLock);
   222→
   223→    if (prevLock != null) {
   224→      await prevLock.future;
   225→    }
   226→
   227→    try {
   228→      return await action();
   229→    } finally {
   230→      myLock.complete();
   231→      _lockQueue.remove(myLock);
   232→    }
   233→  }

   235→  // ═══ FIX: Recover items stuck in "syncing" state from previous crashes/restarts ═══
   236→  Future<void> _recoverStuckSyncingItems() async {
   237→    if (_box == null || !_box!.isOpen) return;
   238→
   239→    final data = _safeBox?.get(_syncQueueKey);
   240→    if (data == null || data.isEmpty) return;
   241→
   242→    try {
   243→      final decoded = jsonDecode(_encryption.decrypt(data));
   244→      final queue = List<Map<String, dynamic>>.from(decoded);
   245→
   246→      int recovered = 0;
   247→      for (int i = 0; i < queue.length; i++) {
   248→        final item = queue[i];
   249→        if (item['_syncing'] == true) {
   250→          // Reset stuck items: remove the _syncing flag, reset retry count
   251→          // so they get picked up on next sync attempt
   252→          queue[i] = Map<String, dynamic>.from(item);
   253→          queue[i].remove('_syncing');
   254→          queue[i]['retry_count'] = (item['retry_count'] ?? 0);
   255→          queue[i]['_recovered'] = true;
   256→          recovered++;
   257→        }
   258→      }
   259→
   260→      if (recovered > 0) {
   261→        final encrypted = _encryption.encrypt(jsonEncode(queue));
   262→        await _safeBox?.put(_syncQueueKey, encrypted);
   263→        if (kDebugMode)
   264→          debugPrint('[OfflineManager] Recovered $recovered stuck syncing items');
   265→      }
   266→    } catch (e) {
   267→      if (kDebugMode) print('[OfflineManager] Recovery check failed: $e');
   268→    }
   269→  }

   271→  // ===== SUBMISSIONS QUEUE =====
   272→  // ═══ FIX O(n): Sharded storage — each item in its own Hive key ═══
   273→  // Previously: entire queue = single encrypted blob → O(n) decrypt+encrypt per operation
   274→  // Now: each item = `sync_queue/$offlineId` (encrypted individually)
   275→  //       index = `sync_queue_index` (plain JSON list of IDs)
   276→  // addToSyncQueue = O(1): encrypt 1 item + append to index
   277→  // removeFromQueue = O(1): delete 1 key + remove from index
   278→  // _getQueue = O(n): read index + decrypt each item (only when needed)
   279→  static const String _syncQueueIndexKey = 'sync_queue_index';
   280→
   281→  /// Add a submission to the offline sync queue with a unique idempotency key.
   282→  Future<String> addToSyncQueue(Map<String, dynamic> submission) async {
   283→    return _withWriteLock(() async {
   284→      final offlineId = _uuid.v4();
   285→      submission['offline_id'] = offlineId;
   286→      submission['idempotency_key'] = offlineId;
   287→      submission['created_at'] = DateTime.now().toIso8601String();
   288→      submission['retry_count'] = 0;
   289→
   290→      // Validate payload size
   291→      final payloadSize = jsonEncode(submission).length;
   292→      if (payloadSize > _maxPayloadSize) {
   293→        throw ValidationException(
   294→          'Submission payload too large (${payloadSize ~/ 1024}KB, max ${_maxPayloadSize ~/ 1024}KB)',
   295→          fieldErrors: {'size': 'exceeds 1MB limit'},
   296→        );
   297→      }
   298→
   299→      // ═══ O(1): Encrypt single item + append to index ═══
   300→      final encrypted = _encryption.encrypt(jsonEncode(submission));
   301→      await _safeBox?.put('sync_queue/$offlineId', encrypted);
   302→
   303→      // Update index
   304→      final index = _getQueueIndex();
   305→      index.add(offlineId);
   306→      await _safeBox?.put(_syncQueueIndexKey, jsonEncode(index));
   307→      _invalidatePendingCount();
   308→
   309→      return offlineId;
   310→    });
   311→  }

   320→  /// Get the queue index (list of offline IDs)

   323→    if (data == null || data.isEmpty) return [];

   331→  /// Read all queue items — decrypts each individually
   332→  List<Map<String, dynamic>> _getQueue() {
   333→    // ═══ FIX: Try sharded format first, fall back to legacy blob ═══
   334→    final index = _getQueueIndex();
   335→    if (index.isNotEmpty) {
   336→      final items = <Map<String, dynamic>>[];
   337→      for (final id in index) {
   338→        final data = _safeBox?.get('sync_queue/$id');
   339→        if (data == null || data.isEmpty) continue;
   340→        try {
   341→          final decrypted = _encryption.decrypt(data);
   342→          items.add(Map<String, dynamic>.from(jsonDecode(decrypted)));
   343→        } catch (e) {
   344→          if (kDebugMode) print('[OfflineManager] Queue item decrypt error ($id): $e');
   345→        }
   346→      }
   347→      return items;
   348→    }
   349→
   350→    // Legacy fallback: single blob format
   351→    final data = _safeBox?.get(_syncQueueKey);
   352→    if (data == null || data.isEmpty) return [];
   353→    try {
   354→      final decoded = jsonDecode(_encryption.decrypt(data));
   355→      final items = List<Map<String, dynamic>>.from(decoded);
   356→      // Migrate to sharded format
   357→      _migrateQueueToSharded(items);
   358→      return items;
   359→    } catch (e) {
   360→      if (kDebugMode) print('[OfflineManager] Queue decrypt error: $e');
   361→      return [];
   362→    }
   363→  }
   364→
   365→  /// Migrate legacy blob queue to sharded format
   366→  Future<void> _migrateQueueToSharded(List<Map<String, dynamic>> items) async {
   367→    try {
   368→      if (kDebugMode) debugPrint('[OfflineManager] Migrating ${items.length} queue items to sharded format');
   369→      final index = <String>[];
   370→      for (final item in items) {
   371→        final id = item['offline_id'] as String? ?? _uuid.v4();
   372→        item['offline_id'] = id;
   373→        final encrypted = _encryption.encrypt(jsonEncode(item));
   374→        await _safeBox?.put('sync_queue/$id', encrypted);
   375→        index.add(id);
   376→      }
   377→      await _safeBox?.put(_syncQueueIndexKey, jsonEncode(index));
   378→      await _safeBox?.delete(_syncQueueKey); // Delete old blob
   379→      if (kDebugMode) debugPrint('[OfflineManager] ✅ Queue migration complete');
   380→    } catch (e) {
   381→      if (kDebugMode) debugPrint('[OfflineManager] Queue migration failed: $e');
   382→    }
   383→  }

   385→  Future<void> _saveQueue(List<Map<String, dynamic>> queue) async {
   386→    // ═══ FIX: Rebuild sharded storage from scratch ═══
   387→    // Delete old keys, write new ones
   388→    final oldIndex = _getQueueIndex();
   389→    final newIndex = <String>[];
   390→
   391→    for (final item in queue) {
   392→      final id = item['offline_id'] as String? ?? _uuid.v4();
   393→      item['offline_id'] = id;
   394→      final encrypted = _encryption.encrypt(jsonEncode(item));
   395→      await _safeBox?.put('sync_queue/$id', encrypted);
   396→      newIndex.add(id);
   397→    }
   398→
   399→    // Delete keys that are no longer in the queue
   400→    final newSet = newIndex.toSet();
   401→    for (final oldId in oldIndex) {
   402→      if (!newSet.contains(oldId)) {
   403→        await _safeBox?.delete('sync_queue/$oldId');
   404→      }
   405→    }
   406→
   407→    await _safeBox?.put(_syncQueueIndexKey, jsonEncode(newIndex));
   408→    // Delete legacy blob if it still exists
   409→    await _safeBox?.delete(_syncQueueKey);
   410→  }

   416→  /// ═══ FIX: removeFromQueue with write lock to prevent race conditions ═══
   417→  /// Previously: no lock → race condition with addToSyncQueue
   418→  ///   Thread A reads queue (5 items), Thread B adds item (6),
   419→  ///   Thread A saves (4 items) → new item lost!
   420→  /// Now: protected by same mutex as addToSyncQueue/syncPendingItems
   421→  Future<void> removeFromQueue(String offlineId) async {
   422→    return _withWriteLock(() async {
   423→      final queue = _getQueue();
   424→      queue.removeWhere((item) => item['offline_id'] == offlineId);
   425→      await _saveQueue(queue);
   426→      _invalidatePendingCount();
   427→    });
   428→  }

   430→  // ═══ PROPOSAL 3: Batch remove — 1× decrypt + 1× encrypt instead of N× ═══
   431→  /// Remove multiple items from the queue in a single operation.
   432→  /// This is critical for sync: previously, removing 50 synced items
   433→  /// required 50× (decrypt + encrypt) = 100 operations on main thread.
   434→  /// Now: 1× (decrypt + encrypt) regardless of batch size.
   435→  Future<void> removeFromQueueBatch(List<String> offlineIds) async {
   436→    if (offlineIds.isEmpty) return;
   437→    return _withWriteLock(() async {
   438→      final queue = _getQueue();
   439→      final idSet = offlineIds.toSet();
   440→      queue.removeWhere((item) => idSet.contains(item['offline_id']));
   441→      await _saveQueue(queue);
   442→      _invalidatePendingCount();
   443→    });
   444→  }

   446→  /// ═══ FIX M-1: Update specific items in the sync queue ═══
   447→  /// Used by SyncService to persist retry_count, last_retry_at, next_retry_at
   448→  /// after a failed sync attempt. Without this, retry_count resets to 0 every cycle.
   449→  Future<void> updateQueueItems(List<Map<String, dynamic>> items) async {
   450→    if (items.isEmpty) return;
   451→    return _withWriteLock(() async {
   452→      final queue = _getQueue();
   453→      final updateMap = <String, Map<String, dynamic>>{};
   454→      for (final item in items) {
   455→        final id = item['offline_id'] as String?;
   456→        if (id != null) updateMap[id] = item;
   457→      }
   458→      for (int i = 0; i < queue.length; i++) {
   459→        final id = queue[i]['offline_id'] as String?;
   460→        if (id != null && updateMap.containsKey(id)) {
   461→          queue[i] = updateMap[id]!;
   462→        }
   463→      }
   464→      await _saveQueue(queue);
   465→    });
   466→  }

   468→  Future<void> clearQueue() async {
   469→    // ⚠️ FIX M8: Clear sharded storage too, not just legacy blob
   470→    // Previously: only deleted _syncQueueKey (old blob) — sharded items remained orphaned
   471→    // Now: delete index + all sharded items + legacy blob
   472→    final index = _getQueueIndex();
   473→    for (final id in index) {
   474→      await _safeBox?.delete('sync_queue/$id');
   475→    }
   476→    await _safeBox?.delete(_syncQueueIndexKey);
   477→    await _safeBox?.delete(_syncQueueKey);
   478→    _invalidatePendingCount();
   479→  }

   493→  /// Force recalculate count (call after queue changes)
   494→  void _invalidatePendingCount() {
   495→    _cachedPendingCount = -1;
   496→    // ═══ FIX ME4: Emit new count to stream ═══
   497→    if (!_pendingCountController.isClosed) {
   498→      _pendingCountController.add(pendingCount);
   499→    }
   500→  }

   502→  /// Sync all pending items with retry logic and conflict handling.
   503→  /// ═══ FIX: Process in-memory, save ONCE at end — prevents data loss on crash ═══
   504→  Future<List<OfflineSyncResult>> syncPendingItems(
   505→    Future<Map<String, dynamic>> Function(Map<String, dynamic>) submitFn,
   506→  ) async {
   507→    // ═══ FIX: Wrap with _withWriteLock to prevent race condition ═══
   508→    // Previously: _getQueue() + _saveQueue() without lock
   509→    // Now: protected by same mutex as addToSyncQueue/removeFromQueue
   510→    return _withWriteLock(() async {
   511→    final pending = _getQueue();
   512→    if (pending.isEmpty) return [];
   513→
   514→    final results = <OfflineSyncResult>[];
   515→    final remaining = <Map<String, dynamic>>[];
   516→    final successfullySynced = <String>[];
   517→
   518→    // ═══ FIX: Mark items as _syncing to prevent duplicate processing ═══
   519→    // If we crash during sync, _recoverStuckSyncingItems will reset them on next init
   520→    for (final item in pending) {
   521→      item['_syncing'] = true;
   522→    }
   523→
   524→    for (final item in pending) {
   525→      try {
   526→        // Add sync metadata
   527→        final payload = Map<String, dynamic>.from(item);
   528→        payload.remove('_syncing');
   529→        payload.remove('_recovered');
   530→        payload['sync_metadata'] = {
   531→          'client_timestamp': DateTime.now().toIso8601String(),
   532→          'app_version': AppConfig.appVersion,
   533→          'retry_count': item['retry_count'] ?? 0,
   534→        };
   535→
   536→        final response = await submitFn(payload);
   537→
   538→        if (response['status'] == 'duplicate') {
   539→          successfullySynced.add(item['offline_id']);
   540→          results.add(
   541→            OfflineSyncResult.duplicate(item['offline_id'], response),
   542→          );
   543→        } else if (response['conflict'] == true) {
   544→          await _saveConflict(item, response);
   545→          successfullySynced.add(item['offline_id']);
   546→          results.add(OfflineSyncResult.conflict(item['offline_id'], response));
   547→        } else if (response['success'] == true) {
   548→          successfullySynced.add(item['offline_id']);
   549→          results.add(OfflineSyncResult.success(item['offline_id'], response));
   550→        } else {
   551→          final retryCount = (item['retry_count'] ?? 0) as int;
   552→          if (retryCount < _maxRetries) {
   553→            item['retry_count'] = retryCount + 1;
   554→            item['last_retry_at'] = DateTime.now().toIso8601String();
   555→            item.remove('_syncing');
   556→            remaining.add(item);
   557→            results.add(
   558→              OfflineSyncResult.error(
   559→                item['offline_id'],
   560→                'Unexpected server response',
   561→              ),
   562→            );
   563→          } else {
   564→            item.remove('_syncing');
   565→            remaining.add(item);
   566→            results.add(
   567→              OfflineSyncResult.error(
   568→                item['offline_id'],
   569→                'Unexpected server response',
   570→              ),
   571→            );
   572→          }
   573→        }
   574→      } on ApiException catch (e) {
   575→        final retryCount = (item['retry_count'] ?? 0) as int;
   576→        if (_isRetryableError(e) && retryCount < _maxRetries) {
   577→          item['retry_count'] = retryCount + 1;
   578→          item['last_retry_at'] = DateTime.now().toIso8601String();
   579→          item.remove('_syncing');
   580→          remaining.add(item);
   581→          results.add(
   582→            OfflineSyncResult.error(
   583→              item['offline_id'],
   584→              'RETRY_${retryCount + 1}/$_maxRetries: ${e.message}',
   585→            ),
   586→          );
   587→        } else {
   588→          await _logSyncError(item, e);
   589→          item.remove('_syncing');
   590→          remaining.add(item); // Keep for manual review
   591→          results.add(OfflineSyncResult.error(item['offline_id'], e.message));
   592→        }
   593→      } catch (e) {
   594→        await _logSyncError(item, e);
   595→        item.remove('_syncing');
   596→        remaining.add(item);
   597→        results.add(OfflineSyncResult.error(item['offline_id'], e.toString()));
   598→      }
   599→    }
   600→
   601→    // ═══ FIX: Save ONCE — remaining items only. No intermediate writes. ═══
   602→    await _saveQueue(remaining);
   603→    _invalidatePendingCount();
   604→
   605→    _logSyncSummary(results);
   606→    return results;
   607→    }); // Close _withWriteLock
   608→  }

   709→  Future<void> saveFailedSubmission(Map<String, dynamic> item, String error) async {
   710→    return _withWriteLock(() async {
   711→      try {
   712→        final failed = _getFailedSubmissions();

   715→        // FIX: تجاوز الحد الأقصى → احذف الأقدم

   717→          String? oldestKey;
   718→          String? oldestTime;
   719→          failed.forEach((k, v) {
   720→            final t = v['failed_at'] as String? ?? '';
   721→            if (oldestTime == null || t.compareTo(oldestTime!) < 0) {

   732→          'failed_at': DateTime.now().toIso8601String(),

   844→  // ===== DRAFTS (SHARDED STORAGE — Proposal 2) =====
   845→  // Each draft is stored in its own Hive key: drafts/$draftId
   846→  // An unencrypted index (drafts_index) stores the list of draft IDs.
   847→  // This means: saveDraft = encrypt 1 item (not all),
   848→  //             removeDraft = delete 1 key (not decrypt+encrypt all),
   849→  //             getDraftFormIds = read index (no decrypt at all!)
   850→
   851→  static const String _draftsIndexKey = 'drafts_index';
   852→
   853→  /// Migrate old blob-format drafts to sharded storage (called once on init)
   854→  // ═══ FIX R-C1: Key source for old format migration ═══
   855→  // The encryption key is needed for PBKDF2 re-derivation of old format data.
   856→  // This is set during init() and used by _migrateDraftsToSharded().
   857→  String _encryptionKeyForMigration = '';
   858→
   859→  Future<void> _migrateDraftsToSharded() async {
   860→    final oldBlob = _box?.get(_draftsKey);
   861→    if (oldBlob == null || oldBlob.isEmpty) return; // No old data
   862→
   863→    // Check if already migrated
   864→    final index = _box?.get(_draftsIndexKey);
   865→    if (index != null) return; // Already migrated
   866→
   867→    try {
   868→      if (kDebugMode) debugPrint('[OfflineManager] Migrating drafts to sharded storage...');
   869→
   870→      // ═══ FIX R-C1: Try normal decrypt first, then old format migration ═══
   871→      // Previously: _encryption.decrypt(oldBlob) threw FormatException for old format
   872→      //   → catch block swallowed error → migration silently failed → drafts lost
   873→      // Now: attempt PBKDF2 re-derivation for old format data in background Isolate
   874→      String decryptedBlob;
   875→      try {
   876→        decryptedBlob = _encryption.decrypt(oldBlob);
   877→      } catch (e) {
   878→        // ═══ Old format detected — try PBKDF2 re-derivation ═══
   879→        if (kDebugMode) debugPrint('[OfflineManager] Old format blob — attempting PBKDF2 migration...');
   880→        final migrated = await EncryptionService.migrateOldFormatToNew(
   881→          oldBlob,
   882→          _encryptionKeyForMigration,
   883→          _encryption,
   884→        );
   885→        if (migrated != null) {
   886→          // Successfully migrated — decrypt with new format
   887→          decryptedBlob = _encryption.decrypt(migrated);
   888→          if (kDebugMode) debugPrint('[OfflineManager] ✅ Old format blob migrated via PBKDF2');
   889→        } else {
   890→          // ═══ FIX R-C1: Don't delete — preserve for manual recovery ═══
   891→          // Previously: silently failed → old blob deleted → data lost forever
   892→          // Now: keep old blob in separate key for recovery
   893→          if (kDebugMode) debugPrint('[OfflineManager] ⚠️ Could not migrate old blob — preserving for recovery');
   894→          await _box?.put('_unmigrated_drafts_blob', oldBlob);
   895→          await _box?.delete(_draftsKey);
   896→          return;
   897→        }
   898→      }
   899→
   900→      final drafts = Map<String, dynamic>.from(jsonDecode(decryptedBlob));
   901→      final draftIds = <String>[];
   902→
   903→      for (final entry in drafts.entries) {
   904→        final draftId = entry.key;
   905→        final value = entry.value as Map<String, dynamic>;
   906→        // Store each draft in its own key
   907→        await _box?.put('drafts/$draftId', _encryption.encrypt(jsonEncode(value)));
   908→        draftIds.add(draftId);
   909→      }
   910→
   911→      // Store the index (unencrypted — just a list of IDs)
   912→      await _box?.put(_draftsIndexKey, _encryption.encrypt(jsonEncode(draftIds)));
   913→
   914→      // Delete old blob
   915→      await _box?.delete(_draftsKey);
   916→
   917→      if (kDebugMode) debugPrint('[OfflineManager] Migrated ${draftIds.length} drafts to sharded storage');
   918→    } catch (e) {
   919→      if (kDebugMode) debugPrint('[OfflineManager] Draft migration failed: $e');
   920→    }
   921→  }

   923→  /// Save a draft — encrypts ONLY this draft (not all drafts)
   924→  Future<void> saveDraft(
   925→      String draftId, String formId, Map<String, dynamic> data) async {
   926→    return _withWriteLock(() async {
   927→      final draftData = {
   928→        'form_id': formId,
   929→        'data': data,

   931→      };
   932→

   935→      //   → can't decrypt on restart → permanent data loss!
   936→      // Now: when _encryptionAvailable is false, save as plain JSON directly.
   937→      String encrypted;
   938→      if (!_encryptionAvailable) {
   939→        debugPrint('[OfflineManager] ⚠️ M-3: Encryption not available — saving as plain JSON');
   940→        encrypted = jsonEncode(draftData);
   941→      } else {
   942→        try {
   943→          final json = jsonEncode(draftData);
   944→          encrypted = _encryption.encrypt(json);
   945→        } catch (e) {
   946→          debugPrint('[OfflineManager] Encrypt failed — saving draft as plain JSON: $e');
   947→          encrypted = jsonEncode(draftData);
   948→        }
   949→      }
   950→
   951→      await _box?.put('drafts/$draftId', encrypted);
   952→
   953→      // Update index
   954→      final indexStr = _box?.get(_draftsIndexKey) ?? '[]';
   955→      List<String> index;
   956→      try {
   957→        // Try encrypted first, then plain JSON
   958→        try {
   959→          index = List<String>.from(jsonDecode(_encryption.decrypt(indexStr)));
   960→        } catch (_) {
   961→          index = List<String>.from(jsonDecode(indexStr));
   962→        }
   963→      } catch (_) {
   964→        index = [];
   965→      }
   966→      if (!index.contains(draftId)) {
   967→        index.add(draftId);
   968→        // ═══ FIX: حفظ الـ index مع fallback ═══
   969→        try {
   970→          await _box?.put(_draftsIndexKey, _encryption.encrypt(jsonEncode(index)));
   971→        } catch (e) {
   972→          // Fallback: plain JSON index
   973→          debugPrint('[OfflineManager] Index encrypt failed — saving as plain JSON: $e');
   974→          await _box?.put(_draftsIndexKey, jsonEncode(index));
   975→        }
   976→      }
   977→    });
   978→  }

  1296→  /// Keys matching these prefixes are reference data — exempt from LRU eviction.

  1300→  Future<void> cacheData(String key, Map<String, dynamic> data) async {
  1301→    return _withWriteLock(() async {
  1302→      final cache = _getCache();
  1303→
  1304→      // ═══ FIX Storage: فحص حجم entry الواحد ═══
  1305→      final entryJson = jsonEncode(data);
  1306→      if (entryJson.length > _maxSingleEntrySize) {
  1307→        if (kDebugMode) {
  1308→          debugPrint('[OfflineManager] Cache entry too large ($key: ${entryJson.length ~/ 1024}KB > ${_maxSingleEntrySize ~/ 1024}KB) — skipping');
  1309→        }
  1310→        return; // لا نخزن entries ضخمة
  1311→      }
  1312→
  1313→      // ═══ FIX: Remove old entry for same key first ═══
  1314→      cache.remove(key);
  1315→
  1316→      // ═══ FIX N-1: LRU eviction — skip reference data (forms, governorates, etc.) ═══
  1317→      // Reference data is small, rarely changes, and MUST be available offline.
  1318→      // Only evict non-reference entries (submissions, analytics, etc.).
  1319→      while (cache.length >= _maxCacheMemoryEntries) {
  1320→        String? oldestKey;
  1321→        DateTime? oldestTime;
  1322→        for (final e in cache.entries) {
  1323→          // Skip reference data — never evict forms, governorates, districts, etc.
  1324→          final isReference = _referenceCachePrefixes.any((prefix) => e.key.startsWith(prefix));
  1325→          if (isReference) continue;
  1326→          final t = DateTime.tryParse(e.value['cached_at'] ?? '');
  1327→          if (t != null && (oldestTime == null || t.isBefore(oldestTime))) {
  1328→            oldestTime = t;
  1329→            oldestKey = e.key;
  1330→          }
  1331→        }
  1332→        if (oldestKey != null) {
  1333→          cache.remove(oldestKey);
  1334→        } else {
  1335→          break; // No non-reference entries to evict
  1336→        }
  1337→      }
  1338→
  1339→      cache[key] = {
  1340→        'data': data,
  1341→        'cached_at': DateTime.now().toIso8601String(),
  1342→      };
  1343→
  1344→      final jsonStr = jsonEncode(cache);
  1345→
  1346→      // ═══ FIX N-1: Total size eviction — skip reference data ═══
  1347→      if (jsonStr.length > _maxTotalCacheSize) {
  1348→        if (kDebugMode) {
  1349→          debugPrint('[OfflineManager] Cache blob too large (${jsonStr.length ~/ 1024}KB) — evicting non-reference entries');
  1350→        }
  1351→        // Remove oldest NON-REFERENCE entries until under limit
  1352→        while (cache.length > 5) {
  1353→          String? oldestKey;
  1354→          DateTime? oldestTime;
  1355→          for (final e in cache.entries) {
  1356→            // Skip reference data
  1357→            final isReference = _referenceCachePrefixes.any((prefix) => e.key.startsWith(prefix));
  1358→            if (isReference) continue;
  1359→            final t = DateTime.tryParse(e.value['cached_at'] ?? '');
  1360→            if (t != null && (oldestTime == null || t.isBefore(oldestTime))) {
  1361→              oldestTime = t;
  1362→              oldestKey = e.key;
  1363→            }
  1364→          }
  1365→          if (oldestKey != null) {
  1366→            cache.remove(oldestKey);
  1367→          } else {
  1368→            break; // No non-reference entries to evict
  1369→          }
  1370→          if (jsonEncode(cache).length < _maxTotalCacheSize ~/ 2) break;
  1371→        }
  1372→      }
  1373→
  1374→      final encrypted = _encryption.encrypt(jsonEncode(cache));
  1375→      await _safeBox?.put(_cacheKey, encrypted);
  1376→      _cacheMemory = cache;
  1377→      _cacheRawSignature = null;
  1378→    });
  1379→  }

  1381→  Map<String, dynamic> _getCache() {
  1382→    // ═══ PERFORMANCE: Return memory cache if available ═══
  1383→    if (_cacheMemory != null) return _cacheMemory!;
  1384→
  1385→    final data = _safeBox?.get(_cacheKey);
  1386→    if (data == null || data.isEmpty) return {};
  1387→
  1388→    // 1. Try decrypting with current key
  1389→    try {
  1390→      final decrypted = _encryption.decrypt(data);
  1391→      final cache = Map<String, dynamic>.from(jsonDecode(decrypted));
  1392→      _cacheMemory = cache; // Cache in memory
  1393→      return cache;
  1394→    } catch (decryptError) {
  1395→      // 2. Decryption failed — try reading as plain JSON
  1396→      //    (handles migration from unencrypted versions)
  1397→      try {
  1398→        final cache = Map<String, dynamic>.from(jsonDecode(data));
  1399→        _cacheMemory = cache;
  1400→        return cache;
  1401→      } catch (_) {
  1402→        // 3. Both failed — data is corrupted, key changed, OR old encryption format
  1403→        // ⚠️ FIX L2: Backup corrupted cache before deleting
  1404→        // Previously: deleted without backup → data lost forever
  1405→        // Now: save to recovery key for potential later recovery
  1406→        if (kDebugMode) {
  1407→          debugPrint(
  1408→            '[OfflineManager] ⚠️ Cache corrupted or old format — backing up before clearing',
  1409→          );
  1410→        }
  1411→        try {
  1412→          // ═══ FIX M-5: synchronous operations to prevent data loss on crash ═══
  1413→          // Previously: fire-and-forget — put and delete ran in parallel
  1414→          // If app crashed before put completed, backup was lost
  1415→          // Note: _getCache() is sync so we cannot await here.
  1416→          // Hive operations are fast local writes — fire-and-forget is acceptable.
  1417→          _safeBox?.put('${_cacheKey}_corrupted_backup', data);
  1418→        } catch (_) {}
  1419→        _safeBox?.delete(_cacheKey);
  1420→        _cacheMemory = null;
  1421→        return {};
  1422→      }
  1423→    }
  1424→  }

  1501→  Future<void> _cleanupOldData() async {

  1503→

  1505→      int removedCount = 0;

  1514→          draftIndex = List<String>.from(jsonDecode(_encryption.decrypt(indexStr)));
  1515→        } catch (_) {
  1516→          draftIndex = List<String>.from(jsonDecode(indexStr));

  1518→      } catch (_) {
  1519→        draftIndex = [];
  1520→      }
  1521→
  1522→      final draftsToRemove = <String>[];
  1523→      final draftsToPreserve = <String, String>{}; // id → raw data (for recovery)
  1524→      // ═══ FIX: Limit PBKDF2 migration attempts to prevent long init times ═══
  1525→      // Each attempt = ~2-3s (PBKDF2 600k in Isolate). Max 5 attempts = ~15s.
  1526→      int migrationAttempts = 0;
  1527→      const maxMigrationAttempts = 5;
  1528→      for (final draftId in draftIndex) {
  1529→        final data = _box!.get('drafts/$draftId');
  1530→        if (data == null || data.isEmpty) {
  1531→          draftsToRemove.add(draftId);
  1532→          continue;
  1533→        }
  1534→        try {
  1535→          String? decrypted;
  1536→          try {
  1537→            decrypted = _encryption.decrypt(data);

  1540→            try {
  1541→              final v = jsonDecode(data);
  1542→              if (v is Map && v.containsKey('form_id') && v.containsKey('data')) {
  1543→                decrypted = data;  // plain JSON صالح — لا تُحذف!
  1544→                if (kDebugMode) debugPrint('[OfflineManager] Draft $draftId is plain JSON — keeping');
  1545→              }
  1546→            } catch (_) {}
  1547→
  1548→            // ═══ محاولة 2: old format migration ═══
  1549→            if (decrypted == null && EncryptionService.isOldFormat(data) && _encryptionKeyForMigration.isNotEmpty && migrationAttempts < maxMigrationAttempts) {

  1551→              migrationAttempts++;
  1552→              final migrated = await EncryptionService.decryptOldFormat(data, _encryptionKeyForMigration);
  1553→              if (migrated != null) {
  1554→                decrypted = migrated;
  1555→                // Re-encrypt with new format for future use
  1556→                try {
  1557→                  final reEncrypted = _encryption.encrypt(migrated);
  1558→                  await _box!.put('drafts/$draftId', reEncrypted);
  1559→                  if (kDebugMode) debugPrint('[OfflineManager] ✅ Draft $draftId migrated to new format');
  1560→                } catch (_) {
  1561→                  // Re-encryption failed, but we have the plaintext
  1562→                }
  1563→              } else {
  1564→                // ═══ FIX R-C1: Don't delete — preserve for manual recovery ═══
  1565→                if (kDebugMode) debugPrint('[OfflineManager] ⚠️ Draft $draftId old format migration failed — preserving');
  1566→                draftsToPreserve[draftId] = data;
  1567→                draftsToRemove.add(draftId); // Remove from active index
  1568→                continue;
  1569→              }
  1570→            } else {
  1571→              // Not old format or no key — preserve for recovery
  1572→              draftsToPreserve[draftId] = data;
  1573→              draftsToRemove.add(draftId);
  1574→              continue;
  1575→            }
  1576→          }
  1577→          if (decrypted == null) {
  1578→            // All decryption attempts failed — preserve for recovery
  1579→            draftsToPreserve[draftId] = data;
  1580→            draftsToRemove.add(draftId);
  1581→            continue;
  1582→          }
  1583→          final draft = jsonDecode(decrypted);
  1584→          final savedAt = DateTime.tryParse(draft['saved_at'] ?? '');
  1585→          if (savedAt != null && now.difference(savedAt).inDays > 30) {
  1586→            draftsToRemove.add(draftId);
  1587→          }
  1588→        } catch (_) {
  1589→          // ═══ FIX R-C1: Don't delete corrupted drafts — preserve for recovery ═══
  1590→          // Previously: draftsToRemove.add(draftId) → deleted forever
  1591→          // Now: preserve raw data in recovery storage
  1592→          draftsToPreserve[draftId] = data;
  1593→          draftsToRemove.add(draftId); // Remove from active index
  1594→        }
  1595→      }
  1596→
  1597→      // ═══ FIX R-C1: Preserve unmigrated/corrupted drafts for manual recovery ═══
  1598→      // FIX: حد أقصى 10 مسودات مستردة لمنع نمو التخزين
  1599→      if (draftsToPreserve.isNotEmpty) {
  1600→        final existingRecovery = _box!.get('_recovered_drafts') ?? '{}';
  1601→        Map<String, dynamic> recoveryMap;
  1602→        try {
  1603→          recoveryMap = Map<String, dynamic>.from(jsonDecode(existingRecovery));
  1604→        } catch (_) {
  1605→          recoveryMap = {};
  1606→        }
  1607→        recoveryMap.addAll(draftsToPreserve);
  1608→        // FIX: احذف الأقدم إذا تجاوز 10
  1609→        while (recoveryMap.length > 10) {
  1610→          recoveryMap.remove(recoveryMap.keys.first);
  1611→        }
  1612→        await _box!.put('_recovered_drafts', jsonEncode(recoveryMap));
  1613→        if (kDebugMode) {
  1614→          debugPrint('[OfflineManager] Preserved ${draftsToPreserve.length} drafts for recovery (total: ${recoveryMap.length})');
  1615→        }
  1616→      }
  1617→
  1618→      for (final id in draftsToRemove) {
  1619→        await _box!.delete('drafts/$id');
  1620→        draftIndex.remove(id);
  1621→        removedCount++;
  1622→      }
  1623→      if (draftsToRemove.isNotEmpty) {
  1624→        try {
  1625→          await _box!.put(_draftsIndexKey, _encryption.encrypt(jsonEncode(draftIndex)));
  1626→        } catch (_) {
  1627→          await _box!.put(_draftsIndexKey, jsonEncode(draftIndex));
  1628→        }
  1629→      }
  1630→
  1631→      // 2. Clean old failed submissions (older than 30 days — was 7)
  1632→      // ═══ FIX F-4: تمديد مدة الاحتفاظ من 7 إلى 30 يوم ═══
  1633→      // Previously: 7 days → users lose failed submissions if they don't open app for a week
  1634→      // Now: 30 days → more time to notice and retry failed submissions
  1635→      final failed = _getFailedSubmissions();
  1636→      final failedToRemove = <String>[];
  1637→      for (final entry in failed.entries) {
  1638→        final failedAt = DateTime.tryParse(entry.value['failed_at'] ?? '');
  1639→        if (failedAt != null && now.difference(failedAt).inDays > 30) {
  1640→          failedToRemove.add(entry.key);
  1641→        }
  1642→      }
  1643→      for (final id in failedToRemove) {
  1644→        failed.remove(id);
  1645→        removedCount++;
  1646→      }
  1647→      if (failedToRemove.isNotEmpty) {
  1648→        final encrypted = _encryption.encrypt(jsonEncode(failed));
  1649→        await _safeBox?.put(_failedSubmissionsKey, encrypted);
  1650→      }
  1651→
  1652→      // 3. Clean old cache entries (older than 30 days)
  1653→      // ═══ FIX N-2: تمديد مدة الاحتفاظ من 7 إلى 30 يوم ═══
  1654→      // Previously: 7 days → users who don't open app for a week lose all forms
  1655→      // Now: 30 days → forms stay cached for a month
  1656→      // Reference data (forms, governorates, etc.) is NEVER cleaned by time.
  1657→      final cache = _getCache();
  1658→      final cacheToRemove = <String>[];
  1659→      for (final entry in cache.entries) {
  1660→        // Skip reference data — never time-expire forms, governorates, etc.
  1661→        final isReference = _referenceCachePrefixes.any((prefix) => entry.key.startsWith(prefix));
  1662→        if (isReference) continue;
  1663→        final cachedAt = DateTime.tryParse(entry.value['cached_at'] ?? '');
  1664→        if (cachedAt != null && now.difference(cachedAt).inDays > 30) {
  1665→          cacheToRemove.add(entry.key);
  1666→        }
  1667→      }
  1668→      for (final key in cacheToRemove) {
  1669→        cache.remove(key);
  1670→        removedCount++;
  1671→      }
  1672→      if (cacheToRemove.isNotEmpty) {
  1673→        final encrypted = _encryption.encrypt(jsonEncode(cache));
  1674→        await _safeBox?.put(_cacheKey, encrypted);
  1675→        _cacheMemory = cache;
  1676→      }
  1677→
  1678→      // 4. Clean old sync queue items (older than 30 days — was 7)
  1679→      // ═══ FIX: تمديد مدة الاحتفاظ من 7 إلى 30 يوم ═══
  1680→      // Previously: 7 days → items stuck for a week get deleted
  1681→      // Now: 30 days → more time for retry before giving up
  1682→      final queueIndex = _getQueueIndex();
  1683→      final queueToRemove = <String>[];
  1684→      for (final id in queueIndex) {
  1685→        final data = _safeBox?.get('sync_queue/$id');
  1686→        if (data == null || data.isEmpty) {
  1687→          queueToRemove.add(id);
  1688→          continue;
  1689→        }
  1690→        try {
  1691→          final decrypted = _encryption.decrypt(data);
  1692→          final item = jsonDecode(decrypted);
  1693→          final createdAt = DateTime.tryParse(item['created_at'] ?? '');
  1694→          if (createdAt != null && now.difference(createdAt).inDays > 30) {
  1695→            queueToRemove.add(id);
  1696→            // Save to failed submissions before removing
  1697→            await saveFailedSubmission(item, 'Auto-cleanup: older than 30 days');
  1698→          }
  1699→        } catch (_) {
  1700→          queueToRemove.add(id); // corrupted item
  1701→        }
  1702→      }
  1703→      for (final id in queueToRemove) {
  1704→        await _safeBox?.delete('sync_queue/$id');
  1705→        queueIndex.remove(id);
  1706→        removedCount++;
  1707→      }
  1708→      if (queueToRemove.isNotEmpty) {
  1709→        await _safeBox?.put(_syncQueueIndexKey, jsonEncode(queueIndex));
  1710→        _invalidatePendingCount();
  1711→      }
  1712→
  1713→      // 5. Check total box size — if > 200MB, clear non-reference cache
  1714→      try {
  1715→        final appDir = await getApplicationDocumentsDirectory().timeout(
  1716→          const Duration(seconds: 3),
  1717→        );
  1718→        final boxFile = File('${appDir.path}/$_boxName.hive');
  1719→        if (await boxFile.exists()) {
  1720→          final sizeBytes = await boxFile.length();
  1721→          if (sizeBytes > _maxBoxSizeBytes) {
  1722→            if (kDebugMode) {
  1723→              debugPrint('[OfflineManager] ⚠️ Hive box is ${sizeBytes ~/ (1024*1024)}MB — clearing non-reference cache');
  1724→            }
  1725→            // ═══ FIX N-3: Selective cache clear — preserve reference data ═══
  1726→            // Previously: clearCache() wiped everything including forms
  1727→            // Now: only clear non-reference entries (submissions, analytics, etc.)
  1728→            final cache = _getCache();
  1729→            final keysToRemove = <String>[];
  1730→            for (final key in cache.keys) {
  1731→              final isReference = _referenceCachePrefixes.any((prefix) => key.startsWith(prefix));
  1732→              if (!isReference) keysToRemove.add(key);
  1733→            }
  1734→            for (final key in keysToRemove) {
  1735→              cache.remove(key);
  1736→            }
  1737→            if (keysToRemove.isNotEmpty) {
  1738→              final encrypted = _encryption.encrypt(jsonEncode(cache));
  1739→              await _safeBox?.put(_cacheKey, encrypted);
  1740→              _cacheMemory = cache;
  1741→              _cacheRawSignature = null;
  1742→            }
  1743→            removedCount++;
  1744→          }
  1745→        }
  1746→      } catch (_) {
  1747→        // Can't check size — skip
  1748→      }
  1749→
  1750→      // ═══ FIX Storage: Hive compaction — يُزيل البيانات الميتة من الملف ═══
  1751→      // FIX: compact دوري كل 7 أيام حتى بدون removal لمنع تراكم البيانات الميتة
  1752→      final lastCompactStr = _box?.get('_last_compact_at');
  1753→      bool shouldCompact = removedCount > 0;
  1754→      if (lastCompactStr != null) {
  1755→        try {
  1756→          final lastCompact = DateTime.parse(lastCompactStr as String);
  1757→          if (DateTime.now().difference(lastCompact).inDays > 7) {
  1758→            shouldCompact = true;
  1759→          }
  1760→        } catch (_) {}
  1761→      } else {
  1762→        shouldCompact = true; // أول مرة
  1763→      }
  1764→
  1765→      if (shouldCompact) {
  1766→        try {
  1767→          await _box!.compact();
  1768→          await _box?.put('_last_compact_at', DateTime.now().toIso8601String());
  1769→          if (kDebugMode) {
  1770→            debugPrint('[OfflineManager] Hive compacted (removedCount=$removedCount)');
  1771→          }
  1772→        } catch (e) {
  1773→          if (kDebugMode) debugPrint('[OfflineManager] Compact failed: $e');
  1774→        }
  1775→      }
  1776→
  1777→      if (kDebugMode && removedCount > 0) {
  1778→        debugPrint('[OfflineManager] 🧹 Cleanup: removed $removedCount old entries');
  1779→      }
  1780→    } catch (e) {
  1781→      if (kDebugMode) {
  1782→        debugPrint('[OfflineManager] Cleanup failed: $e');
  1783→      }
  1784→    }
  1785→  }
