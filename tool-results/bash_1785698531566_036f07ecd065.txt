    38→    await manager.init().timeout(

    42→      },

    51→  // ═══ FIX: Bridge ConnectivityUtils updates to OfflineManager ═══
    52→  StreamSubscription? connSub;
    53→  try {
    54→    connSub = ConnectivityUtils.onConnectivityChanged.listen(

   157→class SubmissionsFilter {

   159→  final String? formId;
   160→  final String? governorateId;
   161→  final String? districtId;
   162→  final String? campaignType;
   163→  final int? campaignRound;

   167→
   168→  const SubmissionsFilter({
   169→    this.status,
   170→    this.formId,
   171→    this.governorateId,
   172→    this.districtId,
   173→    this.campaignType,
   174→    this.campaignRound,
   175→    this.limit = 2000,
   176→    this.offset = 0,
   177→    this.lean = false,
   178→  });
   179→
   180→  @override
   181→  bool operator ==(Object other) =>
   182→      identical(this, other) ||
   183→      other is SubmissionsFilter &&
   184→          runtimeType == other.runtimeType &&
   185→          status == other.status &&
   186→          formId == other.formId &&
   187→          governorateId == other.governorateId &&
   188→          districtId == other.districtId &&
   189→          campaignType == other.campaignType &&
   190→          campaignRound == other.campaignRound &&
   191→          limit == other.limit &&
   192→          offset == other.offset &&
   193→          lean == other.lean;

   195→  @override
   196→  int get hashCode => Object.hash(
   197→        status,
   198→        formId,
   199→        governorateId,
   200→        districtId,
   201→        campaignType,
   202→        campaignRound,
   203→        limit,
   204→        offset,
   205→        lean,
   206→      );
   207→
   208→  String get cacheKey {
   209→    final parts = <String>['submissions'];
   210→    if (lean) parts.add('lean');
   211→    if (campaignType != null) parts.add('camp_$campaignType');
   212→    if (campaignRound != null) parts.add('round_$campaignRound');
   213→    if (formId != null) parts.add('form_$formId');

   281→/// Persisted campaign selection — stored in Supabase profiles table,
   282→/// cached locally in Hive for offline access.
   283→class CampaignNotifier extends StateNotifier<CampaignType> {
   284→  final Ref _ref;
   285→
   286→  CampaignNotifier(this._ref) : super(CampaignType.polioCampaign) {
   287→    _load();
   288→  }
   289→
   290→  Future<void> _load() async {
   291→    try {
   292→      final cache = await _ref.read(offlineDataCacheProvider.future);
   293→      final cached = await cache.getMap('active_campaign', () async {
   294→        final db = _ref.read(databaseServiceProvider);
   295→        final result = await db.getActiveCampaign();
   296→        return {'campaign': result};
   297→      }, maxAge: const Duration(days: 30));
   298→      state = CampaignType.fromString(
   299→        cached['campaign'] as String? ?? 'polio_campaign',
   300→      );
   301→    } catch (_) {
   302→      // Default to polio campaign if loading fails
   303→    }
   304→
   305→    // ═══ Load visibility in background (non-blocking) ═══
   306→    _loadVisibilityInBackground();
   307→  }
   308→
   309→  Future<void> _loadVisibilityInBackground() async {
   310→    try {
   311→      final api = _ref.read(apiClientProvider);
   312→      final result = await api.select('campaign_types', select: 'key, visible');
   313→      final visibilityMap = <String, bool>{};
   314→      for (final row in result) {
   315→        final key = row['key'] as String?;
   316→        final visible = row['visible'] as bool?;
   317→        if (key != null) visibilityMap[key] = visible ?? true;
   318→      }
   319→      await CampaignType.loadVisibility(visibilityMap);
   320→    } catch (_) {
   321→      // Silently fail — default to all visible
   322→    }
   323→  }
   324→
   325→
   326→  Future<void> selectCampaign(CampaignType campaign) async {
   327→    if (campaign == state) return;
   328→    final prev = state;
   329→    state = campaign;
   330→    try {
   331→      if (ConnectivityUtils.isOnline) {
   332→        final db = _ref.read(databaseServiceProvider);
   333→        await db.setActiveCampaign(campaign.value);
   334→      }
   335→      _ref.invalidate(formsProvider);
   336→      _ref.invalidate(dashboardAnalyticsProvider);
   337→    } catch (e) {
   338→      state = prev;
   339→      debugPrint('[CampaignNotifier] Save failed — reverted: $e');
   340→    }
   341→  }
   342→}

   350→/// Persisted campaign round — tracks which round is active for integrated activity.
   351→/// Round 1 = الجولة الأولى, Round 2 = الجولة الثانية, etc.
   352→/// Only relevant when campaignType == integrated_activity.
   353→class CampaignRoundNotifier extends StateNotifier<int> {
   354→  final Ref _ref;
   355→
   356→  CampaignRoundNotifier(this._ref) : super(1) {
   357→    _load();
   358→  }
   359→
   360→  Future<void> _load() async {
   361→    try {
   362→      final cache = await _ref.read(offlineDataCacheProvider.future);
   363→      final cached = await cache.getMap('active_campaign_round', () async {
   364→        final db = _ref.read(databaseServiceProvider);
   365→        final result = await db.getAppSettings(key: 'active_campaign_round');
   366→        return {'round': result['value'] ?? 1};
   367→      }, maxAge: const Duration(days: 30));
   368→      state = (cached['round'] as num?)?.toInt() ?? 1;
   369→    } catch (_) {
   370→      // Default to round 1
   371→    }
   372→  }
   373→
   374→  Future<void> selectRound(int round) async {
   375→    if (round == state || round < 1 || round > 5) return;
   376→    state = round;
   377→    try {
   378→      final db = _ref.read(databaseServiceProvider);
   379→      await db.updateAppSetting('active_campaign_round', round);
   380→
   381→      // Invalidate all providers that depend on campaign round.
   382→      // Family.autoDispose providers (submissionTrend, governorateRanking, shortages)
   383→      // will auto-refresh when their consumers re-read with the new round arg.
   384→      _ref.invalidate(formsProvider);
   385→      _ref.invalidate(dashboardAnalyticsProvider);
   386→      _ref.invalidate(formStatsProvider);
   387→
   388→      if (kDebugMode) {
   389→        debugPrint('[CampaignRoundNotifier] Round changed to $round - Providers invalidated');
   390→      }
   391→    } catch (e) {
   392→      debugPrint('[CampaignRoundNotifier] Save failed: $e');
   393→    }
   394→  }
   395→}

   421→final formsProvider = FutureProvider<List<Map<String, dynamic>>>((ref) async {
   422→  final campaign = ref.watch(campaignProvider);
   423→  final cache = await ref.watch(offlineDataCacheProvider.future);
   424→  final campaignKey = 'forms_' + campaign.value;
   425→
   426→  // OFFLINE-FIRST: Always try cache first, never lose forms
   427→  List<Map<String, dynamic>>? allForms;
   428→
   429→  // 1. Try campaign-specific cache (works online and offline)
   430→  try {
   431→    allForms = await cache.getList(
   432→      campaignKey,
   433→      () => ref.read(databaseServiceProvider).getForms(campaignType: campaign.value),
   434→      maxAge: const Duration(hours: 24),
   435→    );
   436→  } catch (_) {}
   437→
   438→  // 2. Fallback: try forms_all cache (cached by sync service warmup)
   439→  if (allForms == null || allForms.isEmpty) {
   440→    try {
   441→      final fallback = cache.getCachedDataList('forms_all');
   442→      if (fallback != null && fallback.isNotEmpty) allForms = fallback;
   443→    } catch (_) {}
   444→  }
   445→
   446→  // 3. Fallback: try any forms_* key in persistent cache
   447→  if (allForms == null || allForms.isEmpty) {
   448→    try {
   449→      final anyForms = cache.findCachedListByPrefix('forms_');
   450→      if (anyForms != null && anyForms.isNotEmpty) allForms = anyForms;
   451→    } catch (_) {}
   452→  }
   453→
   454→  if (allForms == null || allForms.isEmpty) {
   455→    throw Exception('No forms cached — check connection');
   456→  }
   457→
   458→  // Filter inactive only when online
   459→  if (ConnectivityUtils.isOnline) {
   460→    return allForms.where((f) => f['is_active'] == true).toList();
   461→  }
   462→  return allForms;
   463→});

   500→class FormStats {
   501→  final int drafts;
   502→  final int pending;
   503→  final int submitted;
   504→  final bool submittedLoadError; // ═══ FIX: true if count() failed — UI shows error instead of 0 ═══

   508→final formStatsProvider = FutureProvider.autoDispose<FormStats>((ref) async {
   509→  int drafts = 0, pending = 0, submitted = 0;
   510→
   511→  try {
   512→    // Drafts + pending from local storage (fast)
   513→    final offline = await ref.read(offlineManagerProvider.future).timeout(
   514→          const Duration(seconds: 3),
   515→          onTimeout: () => throw Exception('timeout'),
   516→        );
   517→    pending = offline.pendingCount;
   518→    drafts = offline.getDraftFormIds().length;
   519→  } catch (_) {}
   520→
   521→  try {
   522→    // ═══ PERFORMANCE FIX: Single count query instead of loading 2000 submissions ═══
   523→    final db = ref.read(databaseServiceProvider);
   524→    final campaign = ref.read(campaignProvider);
   525→    final round = ref.read(campaignRoundProvider);
   526→
   527→    // ═══ FIX P2-8: Only 'submitted' is valid — migration 015 removed others ═══
   528→    // Previously: tried 'reviewed', 'approved', 'rejected' which don't exist in enum
   529→    try {
   530→      final count = await db.getSubmissionsCount(
   531→        campaignType: campaign.value,
   532→        campaignRound: round,
   533→        status: 'submitted',
   534→      );
   535→      submitted = count;
   536→    } catch (e) {
   537→      // ═══ FIX: Don't silently swallow — track error so UI can show it ═══
   538→      // Previously: catch (_) {} → Dashboard showed 0 submissions on any error
   539→      // Now: submittedLoadError = true → UI shows "خطأ في التحميل" instead of "0"
   540→      debugPrint('[FormStats] getSubmissionsCount failed: $e');
   541→      submitted = -1; // Sentinel: -1 means "error", not "0 submissions"
   542→    }
   543→  } catch (_) {}
   544→
   545→  return FormStats(drafts: drafts, pending: pending, submitted: submitted == -1 ? 0 : submitted, submittedLoadError: submitted == -1);
   546→});

   786→class AIModelSelection {
   787→  final String provider; // 'groq', 'zai', 'openrouter', 'gemini', 'local'
   788→  final String? model; // specific model id (for OpenRouter)
   789→  final bool autoSelect; // auto-select best provider based on query
   790→
   791→  const AIModelSelection({
   792→    this.provider = 'groq',
   793→    this.model,
   794→    this.autoSelect = true,
   795→  });
   796→
   797→  AIModelSelection copyWith(
   798→      {String? provider, String? model, bool? autoSelect}) {
   799→    return AIModelSelection(
   800→      provider: provider ?? this.provider,
   801→      model: model ?? this.model,
   802→      autoSelect: autoSelect ?? this.autoSelect,
   803→    );
   804→  }
   805→}
   806→
   807→class AIModelSelectionNotifier extends StateNotifier<AIModelSelection> {
   808→  AIModelSelectionNotifier() : super(const AIModelSelection());
   809→
   810→  void selectProvider(String provider) {
   811→    state = state.copyWith(provider: provider);
   812→  }
   813→
   814→  void selectModel(String model) {
   815→    state = state.copyWith(model: model);
   816→  }
   817→
   818→  void toggleAutoSelect() {
   819→    state = state.copyWith(autoSelect: !state.autoSelect);
   820→  }
   821→
   822→  /// Auto-select best provider based on query type
   823→  String selectBestProvider(String query) {
   824→    final intents = EpiNLPEngine.detectIntents(query);
   825→    if (intents.isEmpty) return 'groq';
   826→
   827→    final topIntent = intents.first.intent;
   828→
   829→    // Deep analysis → use powerful model
   830→    if (['generate_report', 'analyze_trend', 'query_analytics']
   831→        .contains(topIntent)) {
   832→      return 'openrouter'; // DeepSeek/GPT-4 for reports
   833→    }
   834→
   835→    // Quick query → use fast model
   836→    if (['query_submissions', 'query_shortages', 'query_governorates']
   837→        .contains(topIntent)) {
   838→      return 'groq'; // Fastest
   839→    }
   840→
   841→    // Knowledge questions → Z AI or Groq
   842→    if (['query_vaccination', 'query_campaign', 'query_aefi', 'query_coverage']
   843→        .contains(topIntent)) {
   844→      return 'zai'; // Z AI good for knowledge
   845→    }
   846→
   847→    // Default → Groq (fastest)
   848→    return 'groq';
   849→  }
   850→}
