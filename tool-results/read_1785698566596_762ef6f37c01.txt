    52→async function getModelConfig(supa: any): Promise<ModelConfig> {
    53→  const now = Date.now()
    54→  if (_modelConfigCache && (now - _modelConfigCache.ts) < MODEL_CONFIG_TTL) {
    55→    return _modelConfigCache.data
    56→  }
    57→  try {
    58→    const { data: model } = await supa.from('ai_models').select('*').eq('is_default', true).eq('is_active', true).single()
    59→    const { data: settings } = await supa.from('app_settings').select('key, value')
    60→      .in('key', ['ai_enabled', 'ai_default_model', 'ai_fallback_enabled', 'ai_stream_enabled', 'ai_max_history', 'ai_rate_limit'])
    61→
    62→    const settingsMap: Record<string, any> = {}
    63→    settings?.forEach((s: any) => { settingsMap[s.key] = s.value })
    64→
    65→    const config: ModelConfig = {
    66→      defaultModel: model,
    67→      enabled: settingsMap.ai_enabled !== false,
    68→      fallbackEnabled: settingsMap.ai_fallback_enabled !== false,
    69→      streamEnabled: settingsMap.ai_stream_enabled !== false,
    70→      maxHistory: Number(settingsMap.ai_max_history) || 20,
    71→      rateLimit: Number(settingsMap.ai_rate_limit) || 25,
    72→    }
    73→    _modelConfigCache = { data: config, ts: now }
    74→    return config
    75→  } catch {
    76→    return { defaultModel: null, enabled: true, fallbackEnabled: true, streamEnabled: true, maxHistory: 20, rateLimit: 25 }
    77→  }
    78→}

    84→async function getUserProfile(supa: any, userId: string): Promise<UserProfile | null> {
    85→  const cached = _userProfileCache.get(userId)
    86→  if (cached && (Date.now() - cached.ts) < PROFILE_CACHE_TTL) return cached.data
    87→
    88→  try {
    89→    const { data } = await supa.from('profiles')
    90→      .select('id, role, full_name, governorate_id, district_id, governorates:governorate_id ( name_ar )')
    91→      .eq('id', userId).single()
    92→
    93→    if (!data) return null
    94→
    95→    const profile: UserProfile = {
    96→      id: data.id,
    97→      role: data.role || 'data_entry',
    98→      full_name: data.full_name || '',
    99→      governorate_id: data.governorate_id,
   100→      district_id: data.district_id,
   101→      governorate_name: data.governorates?.name_ar || null,
   102→    }
   103→    _userProfileCache.set(userId, { data: profile, ts: Date.now() })
   104→    return profile
   105→  } catch {
   106→    return null
   107→  }
   108→}

   114→async function getConversationSummary(supa: any, userId: string): Promise<string> {
   115→  const cached = _summaryCache.get(userId)
   116→  if (cached) return cached
   117→  try {
   118→    const { data } = await supa.from('ai_conversations').select('summary')
   119→      .eq('user_id', userId).order('updated_at', { ascending: false }).limit(1).single()
   120→    if (data?.summary) {
   121→      _summaryCache.set(userId, data.summary)
   122→      return data.summary
   123→    }
   124→  } catch {}
   125→  return ''
   126→}

   154→async function getFeedbackContext(supa: any, userId: string, intent: string): Promise<string> {
   155→  try {
   156→    const { data } = await supa.from('ai_feedback').select('feedback, metadata')
   157→      .eq('user_id', userId).eq('rating', 'down').order('created_at', { ascending: false }).limit(10)
   158→    if (!data?.length) return ''
   159→    const tips = data.filter((f: any) => f.metadata?.intent === intent && f.feedback).map((f: any) => f.feedback).slice(0, 2)
   160→    return tips.length ? `\n⚠️ ملاحظات سابقة: ${tips.join('; ')}. حاول تحسين إجابتك.` : ''
   161→  } catch { return '' }
   162→}

   215→async function fetchLiveData(supa: any, profile: UserProfile | null): Promise<string> {
   216→  // ⚠️ CRITICAL FIX: Was sequential (3+3+5+5+5+5+5=31s worst case).
   217→  // Now: all queries run in parallel (max 5s total instead of 31s)
   218→  const parts: string[] = []
   219→  const isPrivileged = profile && ['admin', 'central', 'governorate'].includes(profile.role)
   220→  const today = todayStart()
   221→
   222→  // Parallel: fetch form IDs
   223→  const [polioFormsRes, integratedFormsRes] = await Promise.all([
   224→    withTimeout(supa.from('forms').select('id').eq('campaign_type', 'polio_campaign').is('deleted_at', null), 3_000).catch(() => null),
   225→    withTimeout(supa.from('forms').select('id').eq('campaign_type', 'integrated_activity').is('deleted_at', null), 3_000).catch(() => null),
   226→  ])
   227→  const polioFormIds = (polioFormsRes?.data || []).map((f: any) => f.id)
   228→  const integratedFormIds = (integratedFormsRes?.data || []).map((f: any) => f.id)
   229→
   230→  // Parallel: all count queries at once
   231→  const queries: Promise<any>[] = []
   232→
   233→  // Today's submissions — polio
   234→  if (polioFormIds.length > 0) {
   235→    let q = supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).in('form_id', polioFormIds).gte('created_at', today)
   236→    if (!isPrivileged && profile) q = q.eq('submitted_by', profile.id)
   237→    queries.push(withTimeout(q, 5_000).then(r => ({ label: '📊 إرساليات اليوم (شلل)', count: r?.count })).catch(() => null))
   238→  }
   239→
   240→  // Today's submissions — integrated
   241→  if (integratedFormIds.length > 0) {
   242→    let q = supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).in('form_id', integratedFormIds).gte('created_at', today)
   243→    if (!isPrivileged && profile) q = q.eq('submitted_by', profile.id)
   244→    queries.push(withTimeout(q, 5_000).then(r => ({ label: '📊 إرساليات اليوم (إيصالي)', count: r?.count })).catch(() => null))
   245→  }
   246→
   247→  // Pending review (privileged only)
   248→  if (isPrivileged) {
   249→    if (polioFormIds.length > 0) {
   250→      queries.push(
   251→        withTimeout(supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).eq('status', 'submitted').in('form_id', polioFormIds), 5_000)
   252→          .then(r => ({ label: '⏳ بانتظار المراجعة (شلل)', count: r?.count })).catch(() => null)
   253→      )
   254→    }
   255→    if (integratedFormIds.length > 0) {
   256→      queries.push(
   257→        withTimeout(supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).eq('status', 'submitted').in('form_id', integratedFormIds), 5_000)
   258→          .then(r => ({ label: '⏳ بانتظار المراجعة (إيصالي)', count: r?.count })).catch(() => null)
   259→      )
   260→    }
   261→    // Active shortages
   262→    queries.push(
   263→      withTimeout(supa.from('supply_shortages').select('severity').is('deleted_at', null).eq('is_resolved', false).limit(2000), 5_000)
   264→        .then(r => {
   265→          if (r?.data?.length) {
   266→            const critical = r.data.filter((s: any) => s.severity === 'critical').length
   267→            return { label: `⚠️ نواقص نشطة: ${r.data.length} (حرجة: ${critical})`, count: r.data.length }
   268→          }
   269→          return null
   270→        }).catch(() => null)
   271→    )
   272→  }
   273→
   274→  // Wait for all queries in parallel (max 5s total)
   275→  const results = await Promise.all(queries)
   276→  for (const r of results) {
   277→    if (r && r.label && r.count != null) parts.push(r.label)
   278→    else if (r && r.label && typeof r.label === 'string' && r.label.startsWith('⚠️')) parts.push(r.label)
   279→  }
   280→
   281→  return parts.join('\n')
   282→}

   286→// ═══════════════════════════════════════════════════════════

   288→async function getSystemHealthScore(supa: any): Promise<any> {
   289→  try {
   290→    const today = todayStart()
   291→    const threeDaysAgo = daysAgo(3)
   292→
   293→    const [todaySubs, pendingSubs, criticalShortages, activeUsers, recentSubs] = await Promise.all([
   294→      withTimeout(supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).gte('created_at', today), 5_000),
   295→      withTimeout(supa.from('form_submissions').select('id', { count: 'exact', head: true }).is('deleted_at', null).eq('status', 'submitted'), 5_000),
   296→      withTimeout(supa.from('supply_shortages').select('id', { count: 'exact', head: true }).is('deleted_at', null).eq('is_resolved', false).eq('severity', 'critical'), 5_000),
   297→      withTimeout(supa.from('profiles').select('id', { count: 'exact', head: true }).eq('is_active', true), 5_000),
   298→      withTimeout(supa.from('form_submissions').select('governorate_id').is('deleted_at', null).gte('created_at', threeDaysAgo).limit(5000), 8_000),
   299→    ])
   300→
   301→    const todayCount = todaySubs?.count || 0
   302→    const pendingCount = pendingSubs?.count || 0
   303→    const criticalCount = criticalShortages?.count || 0
   304→    const userCount = activeUsers?.count || 0
   305→
   306→    const activeGovs = new Set(recentSubs?.map((r: any) => r.governorate_id).filter(Boolean))
   307→    const { data: allGovs } = await supa.from('governorates').select('id').eq('is_active', true)
   308→    const totalGovs = allGovs?.length || 15
   309→    const govActivityRate = Math.round((activeGovs.size / totalGovs) * 100)
   310→
   311→    let score = 100
   312→    let status = '🟢 ممتاز'
   313→    const issues: string[] = []
   314→
   315→    if (todayCount === 0) { score -= 30; issues.push('لا إرساليات اليوم') }
   316→    else if (todayCount < 10) { score -= 15; issues.push(`إرسالات قليلة: ${todayCount}`) }
   317→    if (pendingCount > 50) { score -= 20; issues.push(`${pendingCount} بانتظار المراجعة`) }
   318→    else if (pendingCount > 20) { score -= 10 }
   319→    if (criticalCount > 0) { score -= 25; issues.push(`${criticalCount} نقص حرج`) }
   320→    if (govActivityRate < 50) { score -= 15; issues.push(`${govActivityRate}% نشاط المحافظات`) }
   321→
   322→    if (score < 60) status = '🔴 يحتاج تدخل'
   323→    else if (score < 80) status = '🟡 مقبول'
   324→    else if (score < 90) status = '🟢 جيد'
   325→
   326→    return { score: Math.max(0, score), status, today_submissions: todayCount, pending_review: pendingCount, critical_shortages: criticalCount, active_users: userCount, governorate_activity: govActivityRate + '%', active_governorates: activeGovs.size, total_governorates: totalGovs, issues: issues.length > 0 ? issues : ['كل شي تمام ✅'] }
   327→  } catch { return { score: 0, status: '❌ خطأ في الحساب' } }
   328→}

   377→async function executeFunction(supa: any, name: string, args: Record<string, any>, context?: { campaignRound?: number | null }): Promise<any> {

   386→        let q = supa.from('form_submissions').select('status, governorate_id, created_at, form_id, campaign_round').is('deleted_at', null)
   387→        if (args.status) q = q.eq('status', args.status)
   388→        if (args.days) q = q.gte('created_at', daysAgo(args.days))
   389→        q = applyCampaignFilter(q, formIds, context?.campaignRound)

   404→        ])

   406→      }

   411→        const weekAgo = daysAgo(7)

   413→          withTimeout(supa.from('ai_model_usage').select('tokens_used, success, response_source').gte('created_at', weekAgo).limit(2000), 8_000),
   414→          withTimeout(supa.from('ai_feedback').select('rating, metadata').gte('created_at', weekAgo).limit(100), 5_000),
   415→          withTimeout(supa.from('ai_write_audit').select('tool_name, confirmed_by_user, affected_count').gte('created_at', weekAgo).limit(50), 5_000),
   416→        ])
   417→        const uData = usage?.data || []
   418→        const fData = feedback?.data || []
   419→        const wData = writes?.data || []
   420→        const up = fData.filter((f: any) => f.rating === 'up').length
   421→        const down = fData.filter((f: any) => f.rating === 'down').length
   422→        return {
   423→          period: 'آخر 7 أيام',
   424→          requests: { total: uData.length, successful: uData.filter((u: any) => u.success).length },
   425→          tokens: { total: uData.reduce((s: number, u: any) => s + (u.tokens_used || 0), 0) },
   426→          feedback: { up, down, satisfaction: (up + down) > 0 ? Math.round(up / (up + down) * 100) + '%' : 'N/A' },
   427→          writes: { total: wData.length, confirmed: wData.filter((w: any) => w.confirmed_by_user).length },
   428→        }

   438→          if (batch_current_status) q = q.eq('status', batch_current_status)
   439→          const { count } = await withTimeout(q.select('id', { count: 'exact', head: true }), 15_000) ?? {}
   440→          return { success: true, action: 'batch_update', governorate: batch_governorate, new_status: status, updated_count: count || 0, message: `✅ تم تحديث ${count || 0} إرسالية في ${batch_governorate} إلى "${STATUS_LABELS[status] || status}"` }
   441→        }

   444→        if (err) return { error: `فشل التحديث: ${err.message}` }

   449→        const { title, body, target_role, target_governorate, priority } = args

   451→        let uq = supa.from('profiles').select('id').eq('is_active', true).is('deleted_at', null)
   452→        if (target_role && target_role !== 'all') uq = uq.eq('role', target_role)
   453→        if (target_governorate) {
   454→          const { data: gov } = await supa.from('governorates').select('id').ilike('name_ar', `%${target_governorate}%`).limit(1)
   455→          if (gov?.[0]) uq = uq.eq('governorate_id', gov[0].id)
   456→        }

   458→        if (!users?.length) return { error: 'لا يوجد مستلمين' }
   459→        const notifications = users.map((u: any) => ({ user_id: u.id, title, body, priority: priority || 'normal', is_read: false, created_at: new Date().toISOString() }))
   460→        const { error: err } = await withTimeout(supa.from('notifications').insert(notifications), 15_000)
   461→        if (err) return { error: `فشل: ${err.message}` }
   462→        return { success: true, sent_to: users.length, message: `✅ تم إرسال "${title}" إلى ${users.length} مستخدم` }
   463→      }

   465→      case 'execute_sql': {
   466→        const { query } = args

   469→        if (!normalized.startsWith('SELECT')) return { error: '❌ مسموح فقط باستعلامات SELECT' }
   470→        const forbidden = ['DELETE', 'UPDATE', 'INSERT', 'DROP', 'ALTER', 'TRUNCATE', 'CREATE', 'GRANT', 'REVOKE']

   493→        } else if (data_source === 'submissions_by_day') {

   500→        } else if (data_source === 'users_by_role') {
   501→          // ⚠️ NEW: Users by role chart
   502→          const { data: users } = await withTimeout(supa.from('profiles').select('role').eq('is_active', true).is('deleted_at', null), 5_000) ?? {}
   503→          const roleCounts: Record<string, number> = {}
   504→          users?.forEach((u: any) => { const r = u.role || 'unknown'; roleCounts[r] = (roleCounts[r] || 0) + 1 })
   505→          const roleLabels: Record<string, string> = { admin: 'مدير', central: 'مركزي', governorate: 'مشرف محافظة', district: 'مديرية', data_entry: 'إدخال بيانات' }
   506→          chartData = Object.entries(roleCounts).map(([role, val]) => ({ label: roleLabels[role] || role, value: val }))
   507→        } else if (data_source === 'shortages_by_severity') {

   509→          const { data: shortages } = await withTimeout(supa.from('supply_shortages').select('severity').is('deleted_at', null).eq('is_resolved', false), 5_000) ?? {}

   513→          chartData = Object.entries(sevCounts).map(([sev, val]) => ({ label: sevLabels[sev] || sev, value: val }))

   515→          // ⚠️ NEW: Forms by campaign type chart
   516→          const { data: forms } = await withTimeout(supa.from('forms').select('campaign_type').eq('is_active', true).is('deleted_at', null), 5_000) ?? {}
   517→          const campCounts: Record<string, number> = {}
   518→          forms?.forEach((f: any) => { const ct = f.campaign_type || 'unknown'; campCounts[ct] = (campCounts[ct] || 0) + 1 })
   519→          const campLabels: Record<string, string> = { polio_campaign: 'شلل الأطفال', integrated_activity: 'إيصالي تكاملي' }
   520→          chartData = Object.entries(campCounts).map(([ct, val]) => ({ label: campLabels[ct] || ct, value: val }))
   521→        }
   522→
   523→        return { chart_type, title: args.title || data_source, items: chartData.map((d, i) => ({ ...d, color: CHART_COLORS[i % CHART_COLORS.length] })), generated_at: new Date().toISOString() }
   524→      }

   526→      case 'bulk_export': {
   527→        const { data_type, format, campaign_type, governorate_name, days, limit } = args

   529→        const exportLimit = Math.min(limit || 1000, 5000)
   530→        let q = supa.from(data_type === 'shortages' ? 'supply_shortages' : data_type === 'users' ? 'profiles' : data_type === 'governorates' ? 'governorates' : data_type === 'forms' ? 'forms' : 'form_submissions')
   531→          .select('*').is('deleted_at', null).limit(exportLimit)
   532→        if (data_type !== 'users' && data_type !== 'governorates') q = q.gte('created_at', since)
   533→        const { data } = await withTimeout(q, 15_000) ?? {}
   534→        return { data_type, count: data?.length || 0, records: data || [], export_format: format || 'json', exported_at: new Date().toISOString(), message: `✅ تم تصدير ${data?.length || 0} سجل من ${data_type}` }
   535→      }
   536→
   537→      case 'create_scheduled_report': {
   538→        const { name, report_type, schedule, recipients, campaign_type } = args
   539→        if (!name) return { error: 'اسم التقرير مطلوب' }
   540→        const { data: report, error: err } = await withTimeout(supa.from('scheduled_reports').insert({ name, report_type, schedule: schedule || '0 8 * * *', recipients: recipients || 'admin', campaign_type: campaign_type || 'all', is_active: true, created_at: new Date().toISOString() }).select().single(), 10_000)
   541→        if (err) return { error: `فشل: ${err.message}` }
   542→        return { success: true, report_id: report?.id, message: `✅ تم إنشاء "${name}" — يُرسل ${report_type === 'daily' ? 'يومياً' : report_type === 'weekly' ? 'أسبوعياً' : 'شهرياً'}` }

   544→

   967→          }))
   968→          .sort((a, b) => b.total - a.total)
   969→
   970→        const topPerformers = insights.slice(0, 5)

   974→          total_supervisors: insights.length,
   975→          total_submissions: subs.length,

   980→        }

  1026→async function multiStepToolCalling(
  1027→  msgs: any[], groqKey: string, supa: any,
  1028→  opts: { model: string; maxTokens: number; temperature: number; maxSteps?: number; userId?: string; campaignRound?: number | null }

  1030→  const maxSteps = opts.maxSteps ?? 3
  1031→  const toolCallsUsed: string[] = []
  1032→  let totalTokens = 0
  1033→
  1034→  for (let step = 0; step < maxSteps; step++) {
  1035→    const result = await groqChat(msgs, groqKey, { model: opts.model, maxTokens: opts.maxTokens, temperature: opts.temperature, tools: TOOLS })

  1043→      totalTokens += result.usage?.total_tokens || 0
  1044→    } else if (result.type === 'message') {
  1045→      totalTokens += result.usage?.total_tokens || 0
  1046→      return { content: result.content, toolCallsUsed, totalTokens }
  1047→    }
  1048→  }
  1049→
  1050→  msgs.push({ role: 'user', content: 'قدم الآن الإجابة النهائية بناءً على كل البيانات المجمّعة.' })
  1051→  const final = await groqChat(msgs, groqKey, { model: opts.model, maxTokens: opts.maxTokens, temperature: opts.temperature })
  1052→  if (final?.type === 'message') return { content: final.content, toolCallsUsed, totalTokens: totalTokens + (final.usage?.total_tokens || 0) }
  1053→  return null
  1054→}

  1074→      await send({ type: 'start', message: 'جاري التحليل...' })

  1094→            toolCallsUsed.push(fnName)

  1096→        } else if (result.type === 'message') {

  1103→      const final = await groqChat(msgs, groqKey, { model: opts.model, maxTokens: opts.maxTokens, temperature: opts.temperature })

  1105→      else await send({ type: 'error', message: 'لم أتمكن من توليد إجابة' })
  1106→      await send({ type: 'done' })
  1107→    } catch (e) { await send({ type: 'error', message: String(e) }) }
  1108→    finally { try { await writer.close() } catch {} }
  1109→  })()
  1110→
